library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.NUMERIC_STD.ALL;

entity binary_to_gray is
    port
    ( 
       sw : in STD_LOGIC_VECTOR(7 downto 0);
       led : out STD_LOGIC_VECTOR(7 downto 0);
       led_gray: out STD_LOGIC_VECTOR(7 downto 0);
       clk_ext : in std_logic;
       load : in std_logic;
       clk_int: in std_logic;
       seg : out STD_LOGIC_VECTOR(6 downto 0);
       an  : out STD_LOGIC_VECTOR(3 downto 0)
    );
end binary_to_gray;

architecture Behavioral of binary_to_gray is

    Signal led_int : STD_LOGIC_VECTOR(7 downto 0) := "00000000";
    Signal led_gray_int : STD_LOGIC_VECTOR(7 downto 0) := "00000000";
    Signal refresh:  STD_LOGIC_VECTOR(19 downto 0):= (others=>'0');
    signal sel: STD_LOGIC_VECTOR(1 downto 0);
    signal digit:STD_LOGIC_VECTOR(3 downto 0):= "0000";
    signal digit0:STD_LOGIC_VECTOR(3 downto 0):= "0000";
    signal digit1:STD_LOGIC_VECTOR(3 downto 0):= "0000";
    signal digit2:STD_LOGIC_VECTOR(3 downto 0):= "0000";
    
    
begin

    process (sw)
        variable i :integer;
    begin
        i:= to_integer(unsigned(sw));
        digit0 <= std_logic_vector(to_unsigned(i mod 10,4));
        digit1 <= std_logic_vector(to_unsigned((i/10) mod 10,4));
        digit2 <= std_logic_vector(to_unsigned(i/100,4));     
    end process;
    
    sel<= refresh(3  downto 2); 
    process(clk_int)
        begin
        if rising_edge(clk_int) then
            refresh <= std_logic_vector(unsigned(refresh) + 1);
        end if;
    end process;
    
    process(sel)
    begin
        case sel is
            when "00" => an <= "1110";
            digit <= digit0;
            when "01" => an <= "1101";
            digit <= digit1;
            when "10" => an <= "1011";
            digit <= digit2;
            when "11" => an <= "1111";
            -- sting display - urile
            digit <= "0000"; 
            when others =>
            an <= "1111";
            digit <= "0000";
        end case;
    end process;
    
    process(digit)
    begin
        case digit is
            when "0000" => seg <= "1000000"; -- 0
            when "0001" => seg <= "1111001"; -- 1
            when "0010" => seg <= "0100100"; -- 2
            when "0011" => seg <= "0110000"; -- 3
            when "0100" => seg <= "0011001"; -- 4
            when "0101" => seg <= "0010010"; -- 5
            when "0110" => seg <= "0000010"; -- 6
            when "0111" => seg <= "1111000"; -- 7
            when "1000" => seg <= "0000000"; -- 8
            when "1001" => seg <= "0011000"; -- 9
            when others => seg <= "1111111"; -- eroare(nu ma intereseaza afisarea in hexa)
        end case;
    end process;
    
    led <= led_int;
    led_int <=  sw;
        
    led_gray <= led_gray_int;
        
        process(clk_ext)
        begin
            if rising_edge(clk_ext) then
                    if (load = '1') then
                    
                        led_gray_int(7) <= sw(7);
                        for i in 6 downto 0 loop 
                            led_gray_int(i) <= sw(i) xor sw(i+1);
                        end loop;
                        
                    end if;
                end if;
            
        end process;
